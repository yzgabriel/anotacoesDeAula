//o primeiro bit é o sinal 0 para positivo e 1 para negativo

						// qualquer coisa diferente de zero é um.

  					    //Só existe na plataforma Arduino!!!!

						//representa um caractere na tabela ASCII

						//2,147,482=3,647

{
