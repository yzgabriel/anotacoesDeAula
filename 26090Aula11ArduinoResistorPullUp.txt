const int pinBotao = 7;
const int pinLed = 8;

void setup()
{
  pinMode(pinBotao, OUTPUT);
  pinMode(pinBotao, INPUT_PULLUP);
  Serial.begin(9600);
}

void loop()
{
  bool estadoBotao = digitalRead(pinBotao);
  
  if(estadoBotao == LOW) 
    Serial.println("Botao Solto");
  else 
    Serial.println("Botao Pressionado");
  delay(200); //alivio para o simulador
  
  
  
  
    if(estadoBotao == HIGH)
      digitalWrite(pinLed, LOW);
    else
      digitalWrite(pinLed, HIGH);
  	  delay(200);
 	  digitalWrite(pinLed, LOW);
  	  delay(200);
  
}
